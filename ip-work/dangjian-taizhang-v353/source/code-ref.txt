unsigned char hyddataPoint_QC(hyddataPoint* pthis)
{
unsigned char ret = 0;
if(!pthis) return ret;
double vel = -0.1021 + 4.932 * pthis->diameter - 0.9551 * pow(pthis->diameter, 2) +
0.07934 * pow(pthis->diameter, 3) - 0.002362 * pow(pthis->diameter, 4);
float ratio = 0.4;
if(pthis->vel <= vel * (1 + ratio) && pthis->vel >= vel * (1 - ratio))
ret = 1;
return ret;
}

pthis->ndi = 0;

}
void hyddataSec_Destroy(hyddataSec* pthis)
{
if(!pthis) return;
DLinkList* plist = (DLinkList*)pthis->datalist;
if(plist)
{
plist->MoveFirst(plist);
do
{
hyddataPoint* hd = (hyddataPoint*)plist->GetData(plist, NULL);
if(!hd) continue;
hyddataPoint_Destroy(hd);
} while (plist->MoveNext(plist));
DLinkList_Free(plist);
}
pthis->datalist = NULL;
}

void hyddata_Initialise(hyddata* pthis)
{
if(!pthis) return;
memset(pthis, 0, sizeof(hyddata));
pthis->sec = NULL;
pthis->dtmng = Datetime_Init();
pthis->vara = NULL;
pthis->surf = NULL;
}
void hyddata_DestroySurface(hyddata* pthis)
{
if(pthis->surf)
{
for (int i=0; i<SURF_COUNT; i++)
{
cairo_surface_t* surf = (cairo_surface_t*)pthis->surf[i];
if(!surf) continue;
cairo_surface_destroy(surf);
}
free(pthis->surf);
}
pthis->surf = NULL;
}
void hyddata_Destroy(hyddata* pthis)
{
if(!pthis) return;
if(pthis->dtmng) Datetime_Free(pthis->dtmng);
if(pthis->vara)
{
hyddataVara* va = (hyddataVara*)pthis->vara;
for (int i=0; i<MINUTE_DAY; i++)
{
hyddataVara* hv = &va[i];
hyddataVara_Destroy(hv);
}
free(va);
}
pthis->vara = NULL;
hyddata_DestroySurface(pthis);
}
unsigned char hyddata_Add(void* pthis, hyddataPoint* pdata)
{
unsigned char ret = 0;
if(!pthis || !pdata) return ret;
if(0 == hyddataPoint_QC(pdata)) return ret;
int row = pdata->hour * 60 + pdata->minute;
if(row >= MINUTE_DAY || row < 0) return ret;
if(g_cbprog) g_cbprog(row, MINUTE_DAY);
float ftmp = (pdata->diameter / 0.2f);
int col = floor(ftmp);
//if(abs(ftmp - col) <= 0.001f) col -= 1;
if(col >= MINUTE_SECTON || col < 0) return ret;
hyddata* tmp = (hyddata*)pthis;
void*** plist = tmp->sec;
hyddataSec* hds = (hyddataSec*)plist[row][col];
if(!hds)
{
hds = (hyddataSec*)malloc(sizeof(hyddataSec));
hyddataSec_Initialise(hds);
plist[row][col] = hds;
}
if(1 == pdata->valid)
{
double StNDi = 1.0 * 1000.0 * 1000.0 / (pdata->area * pdata->vel);
hds->ndi += (float)(StNDi / (60 * 0.2));
hds->veli += pdata->vel;
}
DLinkList* qlist = (DLinkList*)hds->datalist;
if(!qlist)
{
qlist = (DLinkList*)malloc(sizeof(DLinkList));
DLinkList_Init(qlist);
hds->datalist = qlist;
}
qlist->Add(qlist, pdata, sizeof(hyddataPoint));
ret = 1;
return ret;
}
void hyddata_BuildAvgVel(hyddata* pthis)
{
if(!pthis || pthis->vara) return;
void*** plist = pthis->sec;
for (int i=0; i<MINUTE_DAY; i++)
{
for (int j=0; j<MINUTE_SECTON; j++)
{
hyddataSec* sec = (hyddataSec*)plist[i][j];
if(!sec || !sec->datalist) continue;;
DLinkList* qlist = (DLinkList*)sec->datalist;
if(qlist->m_Size <= 0) continue;
sec->veli = sec->veli / qlist->m_Size;
}
}
}
unsigned char hyddata_BuildVara(hyddata* pthis)
{
unsigned char ret = 0;
if(!pthis || pthis->vara) return ret;
hyddata_BuildAvgVel(pthis);
hyddataVara* tv = (hyddataVara*)malloc(sizeof(hyddataVara) * MINUTE_DAY);
void*** plist = pthis->sec;
int i = 0;
for (i=0; i<MINUTE_DAY; i++)
{
hyddataVara* pv = &tv[i];
hyddataVara_Initialise(pv);
double RnRtSt = 0, StMn2 = 0, StMn3 = 0, StMn4 = 0, Water = 0, RadarZ = 0;
for (int j=0; j<MINUTE_SECTON; j++)
{
hyddataSec* sec = (hyddataSec*)plist[i][j];
if(!sec) continue;;
double dd = (j + 1) * 0.2  - 0.1;
RnRtSt += 6 * M_PI * sec->veli * sec->ndi * pow(dd, 3) * 0.2 / 10000.0;
StMn2 += sec->ndi * 0.2 * dd * dd;
StMn3 += sec->ndi * 0.2 * dd * dd * dd;
StMn4 += sec->ndi * 0.2 * dd * dd * dd * dd;
Water += M_PI * sec->ndi * 0.2 * dd * dd * dd / (6000.0);
RadarZ += sec->ndi * 0.2 * pow(dd, 6);
}
pv->para[0] = RnRtSt;
pv->StMn2 = StMn2;
pv->StMn3 = StMn3;
pv->StMn4 = StMn4;
if(RadarZ > 0)
pv->RadarZ = 10.0*log10(RadarZ);
if(pv->StMn3 > 0)
{
pv->para[1] = StMn4 / StMn3;
pv->para[2] = log10(256.0 * 1000.0 * Water / (M_PI * pow(pv->para[1], 4)));
}
if(pv->RadarZ >= 0)
pv->para[3] = pv->RadarZ;
}
pthis->vara = tv;
return ret;
}

sprintf(f2, "%s%s_ndi.csv", path, filename);
FILE* fpv = fopen(f1, "wt");
FILE* fp = fopen(f2, "wt");
if(fpv && fp)
{
hyddata_ExportTitleImage(tmp, fpv, fp);
void*** plist = tmp->sec;
char line[1024] = {0}, strtmp[16] = {0};
for (int i=0; i<MINUTE_DAY; i++)
{
int hour = i / 60;
int minute = i % 60;
memset(line, 0, 1024);
sprintf(line, "%d,%d", hour, minute);
for (int j=0; j<MINUTE_SECTON; j++)
{
hyddataSec* hds = (hyddataSec*)plist[i][j];
if(!hds) sprintf(strtmp, ",0");
else sprintf(strtmp, ",%.4f", hds->ndi);
strcat(line, strtmp);
}
strcat(line, "\n");
fputs(line, fp);
hyddataVara* dv = &hv[i];
sprintf(line, "%d,%d,%f,%f,%f,%f\n", hour, minute, dv->para[0], dv->para[1], dv->para[2], dv->para[3]);
fputs(line, fpv);
ret = 1;
}
}
if(fp)	fclose(fp);
if(fpv) fclose(fpv);
return ret;
}

void* hyddata_Init(char* datafile, char* config /*= "hyd2asc.set"*/, CallBackMessage cbmsg /*= NULL*/, CallBackProgress cbprog /*= NULL*/)
{
hyddata* ret = NULL;
if(!datafile) return ret;
ret = (hyddata*)malloc(sizeof(hyddata));
hyddata_Initialise(ret);
g_cbprog = cbprog;
g_cbmsg = cbmsg;
void*** tmp = NULL;
tmp = (void***)malloc(sizeof(void**) * MINUTE_DAY);
for (int i=0; i<MINUTE_DAY; i++)
{
tmp[i] = (void**)malloc(sizeof(void*) * MINUTE_SECTON);
for (int j=0; j<MINUTE_SECTON; j++) tmp[i][j] = NULL;
}
ret->sec = tmp;
char* pconf = config;
if(!pconf) pconf = "hyd2asc.set";
if(0 == hyddata_Read(ret, MINUTE_DAY, MINUTE_SECTON, datafile, pconf, hyddata_Add))
{
hyddata_Free(ret);
ret = NULL;
}
else
{
Datetime* dt = (Datetime*)ret->dtmng;
time_t t = time(NULL);
char str[8] = "YYYY";
dt->Format(dt, str, t);
char filename[1024]; sprintf(filename, "%s", datafile);Charset_ReplaceStrSelf(filename, "\\", "/");
char* p = strrchr(filename, '/');
if(!p) p = filename;
else p += 1;
memcpy(&str[2], &p[1], 2);
int year = atoi(str);
memset(str, 0, 8); memcpy(str, &p[3], 3);
int day = atoi(str);
t = dt->GetByDay(dt, year, day);
sprintf(ret->date, "{yyyy}-{MM}-{DD}");
dt->Replace(dt, ret->date, t);
hyddata_BuildVara(ret);
}
return ret;
}
void hyddata_ImageLabelX(cairo_t* cr, int width, int height, int left, int right, int bottom, int top)
{
char label[64] = "00:00";
cairo_text_extents_t fs;
Cairo_GetTextExtens(cr, &fs, "Sans", 18, label);
int fcn = floor((width - left - right) * 1.0f / (fs.width + 5));
if(fcn > DIM_SECTON_X) fcn = DIM_SECTON_X;
int stepf = ceil(DIM_SECTON_X * 1.0f / fcn);
PointD p;
float stepx = (width - left - right) * 1.0f / (DIM_SECTON_X);
for (int i=0; i<=DIM_SECTON_X; i+=stepf)
{
int hour = i % DIM_SECTON_X;
sprintf(label, "%02d:00", hour);
Cairo_GetTextExtens(cr, &fs, "Sans", 18, label);
p.x = (left + i * stepx) - fs.width / 2.0f + 2;
p.y = height - bottom + 10 + fs.height;
Cairo_DrawString(cr, &p, label);
}
sprintf(label, "Time(hr)");
Cairo_GetTextExtens(cr, &fs, "Sans", 20, label);
p.x = left + (width - left - right - fs.width) / 2.0f;
p.y += fs.height + 10;
Cairo_DrawString(cr, &p, label);
}
int hyddata_ImageLabelY(cairo_t* cr, char* yAxisTitle, int yTickLabelCn, int bits, float* pfmax, float* pfmin, int width, int height, int left, int right, int bottom, int top)
{
if(!cr || !yAxisTitle || !pfmax || !pfmin) return yTickLabelCn;
float fmax = *pfmax;
float fmin = *pfmin;
char bufmin[64] = {0}, bufmax[64] = {0}, ibufmin[32] = {0}, ibufmax[32] = {0}, format[32] = {0};
if(0 == bits)
{
strcpy(format, "%d");
sprintf(bufmin, format, (int)fmin);
sprintf(bufmax, format, (int)fmax);
}
else if(bits > 0)
{
sprintf(format, "%%.%df", bits);
sprintf(bufmin, format, fmin);
sprintf(bufmax, format, fmax);
}
char utf8[128] = {0}, utfi[64] = {0};
sprintf(utf8, "%s", bufmax);
if(strlen(bufmin) > strlen(bufmax)) sprintf(utf8, "%s", bufmin);
float diff = fmax - fmin;
int idf = ceil(diff);
if(0 != idf % 2) diff = (idf / 2 + 1) * 2;
float stepy_val = diff / yTickLabelCn;
if(bits == 0)
{
if(stepy_val < 1)
{
stepy_val = 1;
yTickLabelCn = (int)((fmax - fmin) / stepy_val);
}
else if(floor(stepy_val) != ceil(stepy_val))
{
do
{
yTickLabelCn -= 1;
stepy_val = (fmax - fmin) / yTickLabelCn;
} while (floor(stepy_val) != ceil(stepy_val));
}
}
cairo_text_extents_t fs;
Cairo_GetTextExtens(cr, &fs, "Sans", 18, utf8);
int fcn = floor((height - top - bottom) * 1.0f / (fs.height + 5));
if(fcn > yTickLabelCn) fcn = yTickLabelCn;
int stepf = ceil(yTickLabelCn * 1.0f / fcn);
float stepy = (height - bottom - top) * 1.0 / yTickLabelCn;
PointD p;
for (int i=0; i<= yTickLabelCn; i+=stepf)
{
p.x = left - 15 - fs.width;
p.y = (height - bottom  - i * stepy) + fs.height / 2.0f;
if(0 == bits) sprintf(utf8, "%d", (int)(i * stepy_val + fmin));
else printf(utf8, format, i * stepy_val + fmin);
Cairo_DrawString(cr, &p, utf8);
}
sprintf(utf8, "%s", yAxisTitle);
Cairo_GetTextExtens(cr, &fs, "Sans", 20, utf8);
p.x -= (fs.height - 8);//left - 15 - 10 - fs.height;
p.y = height - bottom - (height - top - bottom - fs.width) / 2.0f;
cairo_move_to(cr, p.x, p.y);
cairo_save(cr);
cairo_rotate(cr, -1 * M_PI / 2.0f);
cairo_show_text(cr, utf8);
cairo_restore(cr);

}

void hyddata_ImageData(void* pthis, cairo_t* cr, int width, int height, int left, int right, int bottom, int top, ColorTable* ct)
{
if(!pthis || !cr || !ct) return;
hyddata* tmp = (hyddata*)pthis;
void*** plist = tmp->sec;
double stepx = (width - left - right) * 1.0f / (MINUTE_DAY);
double stepy = (height - top - bottom) * 1.0f / (MINUTE_SECTON);
double fmin = 10000;
double fmax = 0;
Color c; Color_Default(&c); c.a = 0;
Color* pc = &c;
for (int i=0; i<MINUTE_DAY; i++)
{
int count = 0;
for (int j=0; j<MINUTE_SECTON; j++)
{
hyddataSec* hds = (hyddataSec*)plist[i][j];
float ftmp = 0;
if(hds) ftmp = hds->ndi;
ftmp = hyddata_DataCountTras(ftmp);
if(ftmp < fmin) fmin = ftmp;
if(ftmp > fmax) fmax = ftmp;
pc = ColorTable_FindColor(ct, ftmp);
if(!pc) pc = &c;
cairo_set_source_rgba(cr, pc->r / 255.0f, pc->g / 255.0f, pc->b / 255.0f, pc->a / 255.0f);
cairo_rectangle(cr, left + i*stepx, height - bottom - (j + 1) * stepy, stepx, stepy);
cairo_fill(cr);
}
}
}

unsigned char hyddata_ImageVaraData(cairo_t* cr, float* data, float fmin, float fmax, float unValid, int width, int height, int left, int right, int bottom, int top)
{
unsigned char ret = 0;
if(!cr || !data) return ret;
double sx, sy, dh = height - top - bottom;
double stepx = (width - left - right) *  1.0f / (MINUTE_DAY);
unsigned char first = 0;
int i = 0, j = 0;
do
{
if(fabs(data[i] - unValid) <= DIFF)
{
i += 1;
continue;
}
sx = left + i * stepx;
sy = top + dh - (data[i] - fmin) / (fmax - fmin) * dh;
cairo_move_to(cr, sx, sy);
for (j=i+1; j<MINUTE_DAY; j++)
{
if(fabs(data[j] - unValid) <= DIFF) break;
sx = left + j * stepx;
sy = top + dh - (data[j] - fmin) / (fmax - fmin) * dh;
cairo_line_to(cr, sx, sy);
}
cairo_stroke(cr);
i = j + 1;
}while (i < MINUTE_DAY);
ret = 1;
return ret;
}

namespace hydraw

[UnmanagedFunctionPointer(CallingConvention.Cdecl)]
public delegate void CallbackMessage(string message);
[UnmanagedFunctionPointer(CallingConvention.Cdecl)]
public delegate void CallbackProgress(int index, int count);
public enum hydimg
{
HIT_INIT = 1,
HIT_DM = 2,
HIT_NW = 4,
HIT_RR = 8,
HIT_DBZ=16
};
public unsafe class hyd
{
public static Image GetImage(void* mng, int index)
{
Image img = null;
byte* content = null;
int size = DrawImage_Get(mng, &content, index);
if (size > 0 && null != content)
{
img = BytesToImage(content, size);
}
if (null != content) FreeImageContent(content);
return img;
}
public static Image BytesToImage(byte* content, int size)
{
Image img = null;
byte[] by = new byte[size];
for (int i = 0; i < size; i++) by[i] = content[i];
MemoryStream ms = new MemoryStream(by);
img = Bitmap.FromStream(ms);
ms.Close();
ms.Dispose();
return img;
}
[DllImport("hyd.dll", EntryPoint = "hyddata_Init", CallingConvention = CallingConvention.Cdecl)]
public static extern void* hyddata_Init(string datafile, string config, CallbackMessage cbmsg, CallbackProgress cbprog);
[DllImport("hyd.dll", EntryPoint = "hyddata_Free")]
public static extern void hyddata_Free(void* mng);
[DllImport("hyd.dll", EntryPoint = "hyddata_Export")]
public static extern byte hyddata_Export(void* mng, string filename);
[DllImport("hyd.dll", EntryPoint = "hyddata_ExportData")]
public static extern byte hyddata_ExportData(void* mng, string path, string filename);
[DllImport("hyd.dll", EntryPoint = "hyddata_Image", CallingConvention = CallingConvention.Cdecl)]
public static extern byte hyddata_Image(void* mng, int width, int height, int imgtype, CallbackMessage cbmsg, CallbackProgress cbprog);
[DllImport("hyd.dll", EntryPoint = "hyddata_GetImage")]
public static extern int DrawImage_Get(void* di, byte** img, int index);
[DllImport("hyd.dll", EntryPoint = "FreeImageContent")]
public static extern int FreeImageContent(byte* img);
}
}
using System;
using System.Collections.Generic;
using System.Text;
using System.Drawing;
using System.Windows.Forms;
namespace hydraw
{
public unsafe class hydFile
{
public string filename;
public string name;
public void* mng;
public hydFile()
{
filename ="";
name = "";
mng = null;
}
public void Load()
{
if (null != mng) return;
mng = hyd.hyddata_Init(filename, null, StatusMng.SetMessage, StatusMng.SetProgress);
}
public bool Draw(int width, int height)
{
bool ret = false;
if (true == String.IsNullOrEmpty(filename)) return ret;
if (null == mng) mng = hyd.hyddata_Init(filename, null, StatusMng.SetMessage, StatusMng.SetProgress);
if (null == mng)
{
StatusMng.SetMessage("加载失败：" + filename);
return ret;
}
int type = (int)(hydimg.HIT_INIT | hydimg.HIT_RR | hydimg.HIT_DM | hydimg.HIT_NW | hydimg.HIT_DBZ);
byte by = hyd.hyddata_Image(mng, width, height, type, StatusMng.SetMessage, StatusMng.SetProgress);
if (1 == by)
{
ret = true;
}
else
{
StatusMng.SetMessage("绘图失败：" + filename);
}
return ret;
}
public Image GetImage(int index)
{
Image img = hyd.GetImage(mng, index);
return img;
}
public void Free()
{
if (null == mng) return;
hyd.hyddata_Free(mng);
mng = null;
}
public void Export()
{
if (null == mng)
{
MessageBox.Show("还没加载数据，请先加载数据", "信息提示");
return;
}
SaveFileDialog fd = new SaveFileDialog();
fd.Filter = "Csv文件(*.csv)|*.csv|所有文件(*.*)|*.*";
fd.Title = "导出文件";
fd.FileName = name;
fd.RestoreDirectory = false;
if (fd.ShowDialog() == DialogResult.OK)
{
byte ret = hyd.hyddata_Export(mng, fd.FileName);
if (1 == ret)
StatusMng.SetMessage("导出成功");
else
StatusMng.SetMessage("导出失败");
}
}
public void ExportImageData()
{
if (null == mng)
{
MessageBox.Show("还没加载数据，请先加载数据", "信息提示");
return;
}
SaveFileDialog fd = new SaveFileDialog();
fd.RestoreDirectory = false;
fd.AddExtension = false;
fd.FilterIndex = 1;
fd.FileName = name;
fd.Filter = "CSV文件(*.csv)|*.csv";
if (fd.ShowDialog() == DialogResult.OK)
{
string filename = System.IO.Path.GetFileNameWithoutExtension(fd.FileName);
string path = System.IO.Path.GetDirectoryName(fd.FileName);
byte ret = hyd.hyddata_ExportData(mng, path+"\\", filename);
if (1 == ret)
StatusMng.SetMessage("导出成功");
else
StatusMng.SetMessage("导出失败");
}
}
}
